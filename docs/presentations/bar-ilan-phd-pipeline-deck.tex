% ─── Slide 4 — The pipeline: read · enrich · review · publish ──
% AUDIT: dropped "Stage 1/2/3/4" labels in slide title row per Rule 53;
% kept the four named phases as the only visible labels.
\begin{frame}[t]{Read \textbullet{} Enrich \textbullet{} Review \textbullet{} Publish}
  \begin{itemize}
    \item \textbf{Read.} MARC coded fields \emph{and} free-text notes --- neither dropped.
    \item \textbf{Enrich.} People, places, works linked against Mazal, VIAF, Wikidata, KIMA. This is where most new claims are made. NER and classifiers feed reliable entities into this phase; they don't replace it.
    \item \textbf{Review.} The researcher sees every claim in a workbench. See it, compare it to the evidence, edit it, approve it.
    \item \textbf{Publish.} Three linked representations: full HMO RDF graph, a project Wikibase, and public Wikidata --- one consistent source.
  \end{itemize}
  \vfill
  {\footnotesize Between every phase: a JSON the researcher can edit before the next phase runs.}
\end{frame}

% ─── Slide 6 — Distant supervision ─────────────────────────────
\begin{frame}[t]{Catalog $\rightarrow$ training data, no manual labels}
  \textbf{Distant supervision in three steps}
  \begin{enumerate}
    \item \textbf{Structured MARC.} Names, genres, contents cues from coded fields.
    \item \textbf{Match in free text.} Find the same value inside the catalog's free-text notes.
    \item \textbf{Training pair.} The aligned span becomes a labelled example.
  \end{enumerate}
  \vfill
  {\footnotesize A reusable technique --- any MARC catalog with structured + free-text fields can train its own models.}
\end{frame}
\note{\begin{hebrew}
לפני שנמשיך לסוקר את שאר השכבות, מילה אחת על הצד החישובי. שאלה נפוצה היא איך בכלל מאמנים מודלים כאלה בלי לתייג ידנית אלפי רשומות.
\par
הפתרון הוא להשתמש בקטלוג של הספרייה הלאומית עצמו כמקור לנתוני אימון. בחלק מהרשומות יש מידע מובנה יחסית --- שמות ותפקידים, סימוני תוכן, ז'אנרים, או הערות בעלות --- והוא משמש כעוגן. אחר כך מחפשים את אותו מידע בתוך הטקסט החופשי. אם שם שמופיע בשדה מובנה מופיע גם בהערה חופשית, אפשר ליצור ממנו דוגמת אימון. השיטה הזאת נקראת \textenglish{distant supervision}.
\par
זאת תרומה מתודולוגית עצמאית: כל קטלוג MARC שבו יש שדות מובְנים לצד שדות חופשיים יכול לאמן בו את אותם מודלים, גם בלי לתייג ידנית. \emph{בשקף הבא נראה איפה הם נופלים בסדר התרומה האמיתי.}
\end{hebrew}}

\begin{frame}[t]{Wikidata Studio + the eval-agent}
  \textbf{The workbench}
  \begin{itemize}
    \item Three-axis review: \emph{source} (which MARC field or model), \emph{entity type}, \emph{role} --- the cataloguer's own vocabulary.
    \item Biodata-comparison dialog --- dates, places, name variants, occupations side by side from MARC, Mazal, VIAF, KIMA.
    \item Confidence tooltip --- which sources matched, which guards fired, which biographical years were pulled.
    \item 14 pre-save checks, resumable review state, paging at 25/50/100 rows.
  \end{itemize}
  \textbf{The eval-agent}
  \begin{itemize}
    \item A Gemini-based judge audits \emph{both} entity extraction \emph{and} authority matches.
    \item One shared approval across every editor --- approve once, reflected everywhere.
    \item \textbf{Auto-fix} suggestions for NER entities: when Gemini proposes a high-confidence correction (e.g.\ a missing patronymic), an Auto-fix button appears inline --- text is updated but approval stays human.
  \end{itemize}
  \vfill
  {\footnotesize Studio reviews per claim; the eval-agent audits the corpus and may propose a fix. The agent suggests --- the human decides.}
\end{frame}

\begin{frame}[t]{MARC becomes Linked Open Data, in three layers}
